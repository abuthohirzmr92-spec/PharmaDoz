### 4. Draft → PurchaseItem Mapping

```ts
function draftToPurchaseItem(draftItem: PurchaseDraftItem): PurchaseItem {
  return {
    id: draftItem.id,
    tenantId: "",                           // filled by addPurchase()
    productId: draftItem.matchedProductId!,
    productName: draftItem.rawProductName,
    batchNumber: draftItem.batchNumber ?? "", // auto-gen if empty
    expiredDate: draftItem.expiredDate ?? "",
    quantity: draftItem.quantity,
    unitPrice: draftItem.enteredBuyPrice,
    sellingPrice: draftItem.currentSellingPrice,
  };
}
```
