## STEP 2 — NEW ENTITIES DESIGN

### 2.1 — purchase_drafts

```sql
-- Conceptual schema (DO NOT EXECUTE)
CREATE TABLE purchase_drafts (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id       UUID NOT NULL REFERENCES tenants(id),
  branch_id       UUID REFERENCES branches(id),        -- NULL = all branches

  -- Supplier
  supplier_id     UUID REFERENCES suppliers(id),
  supplier_name   TEXT,                                 -- denormalized for display

  -- Source
  source_type     TEXT NOT NULL CHECK (source_type IN ('manual','excel','csv','ocr')),
  source_reference TEXT,                                -- file name / OCR session ID

  -- Invoice
  invoice_number  TEXT,                                 -- NULL → auto-generate
  purchase_date   TIMESTAMPTZ NOT NULL DEFAULT now(),
  due_date        DATE,

  -- Financial
  subtotal        NUMERIC DEFAULT 0,
  tax_amount      NUMERIC DEFAULT 0,
  grand_total     NUMERIC DEFAULT 0,

  -- Status
  status          TEXT NOT NULL DEFAULT 'draft'
                  CHECK (status IN ('draft','reviewed','confirmed','cancelled')),

  -- OCR linkage
  ocr_session_id  UUID,                                 -- FK to ocr_sessions (future)

  -- Audit
  created_by      UUID REFERENCES profiles(id),
  confirmed_by    UUID REFERENCES profiles(id),
  confirmed_at    TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_purchase_drafts_tenant   ON purchase_drafts(tenant_id);
CREATE INDEX idx_purchase_drafts_branch   ON purchase_drafts(branch_id);
CREATE INDEX idx_purchase_drafts_status   ON purchase_drafts(status);
CREATE INDEX idx_purchase_drafts_source   ON purchase_drafts(source_type);
```

### 2.2 — purchase_draft_items

```sql
CREATE TABLE purchase_draft_items (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  draft_id          UUID NOT NULL REFERENCES purchase_drafts(id) ON DELETE CASCADE,

  -- Product reference
  product_id        UUID REFERENCES products(id),      -- NULL = unmatched
  product_name_raw  TEXT NOT NULL,                      -- original input name (before matching)

  -- Matching
  match_status      TEXT NOT NULL DEFAULT 'unmatched'
                    CHECK (match_status IN ('exact','fuzzy','manual','new_product','unmatched')),
  confidence_score  SMALLINT DEFAULT 0 CHECK (confidence_score BETWEEN 0 AND 100),

  -- Quantity
  quantity          NUMERIC NOT NULL DEFAULT 1 CHECK (quantity > 0),
  unit              TEXT DEFAULT 'pcs',

  -- Pricing
  buy_price         NUMERIC NOT NULL DEFAULT 0,
  previous_buy_price NUMERIC,                           -- auto-loaded from purchase history

  -- Tax
  tax_percent       NUMERIC DEFAULT 0,
  tax_amount        NUMERIC DEFAULT 0,
  tax_mode          TEXT DEFAULT 'exclusive' CHECK (tax_mode IN ('inclusive','exclusive')),

  -- HPP
  hpp               NUMERIC DEFAULT 0,                  -- buy_price + tax_amount

  -- Margin + Recommendation
  target_margin     NUMERIC DEFAULT 0,                  -- from product master
  recommended_sell  NUMERIC DEFAULT 0,                  -- hpp × (1 + margin)
  current_sell      NUMERIC DEFAULT 0,                  -- from product master

  -- Batch
  batch_number      TEXT,                               -- NULL → auto-generate
  expired_date      DATE,

  -- Source tracking
  source_type       TEXT NOT NULL DEFAULT 'manual'
                    CHECK (source_type IN ('manual','excel','csv','ocr')),
  source_reference  TEXT,                               -- row number in Excel / OCR item index

  -- Notes
  notes             TEXT,                               -- manual notes
  warnings          JSONB DEFAULT '[]',                 -- ["Price +15%","Low confidence match"]

  created_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_draft_items_draft    ON purchase_draft_items(draft_id);
CREATE INDEX idx_draft_items_product  ON purchase_draft_items(product_id);
CREATE INDEX idx_draft_items_match    ON purchase_draft_items(match_status);
```

## STEP 3 — OCR SUPPORT FIELDS (FUTURE TABLE)

```sql
-- DO NOT CREATE NOW — for future Phase 4-5

CREATE TABLE ocr_sessions (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id       UUID NOT NULL,
  draft_id        UUID REFERENCES purchase_drafts(id),

  -- File
  file_name       TEXT NOT NULL,

  ocr_processed_at TIMESTAMPTZ,
  ocr_cost        NUMERIC DEFAULT 0,

  -- AI Parser

  ai_confidence   NUMERIC,
  ai_processed_at TIMESTAMPTZ,
  ai_cost         NUMERIC DEFAULT 0,

  processing_status TEXT NOT NULL DEFAULT 'pending'
                   CHECK (processing_status IN ('pending','processing','success','failed')),
  error_message   TEXT,

  -- Dedup
  duplicate_of    UUID REFERENCES ocr_sessions(id),      -- link to first upload

  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_ocr_sessions_tenant ON ocr_sessions(tenant_id);
CREATE INDEX idx_ocr_sessions_hash   ON ocr_sessions(image_hash);
```

### Dependency Map

| New Entity | Depends On | Blocked By |
|-----------|------------|------------|
| `purchase_drafts` | `tenants`, `branches`, `suppliers`, `profiles` | None |
| `purchase_draft_items` | `purchase_drafts`, `products` | None |
| `ocr_sessions` | `purchase_drafts` (optional) | OCR provider integration |

### Risk Assessment

| Risk | Impact | Mitigation |
|------|--------|------------|
| Draft table grows unbounded | LOW | Auto-cleanup: DELETE drafts WHERE status='cancelled' AND created_at < NOW() - INTERVAL '30 days' |
| FK violation on confirm (product deleted) | MEDIUM | Validate all product_id references before confirming draft |
| Branch mismatch (draft from branch A, user in branch B) | MEDIUM | Draft inherits branch_id from active branch at creation time |
| Duplicate draft from same source | LOW | source_type + source_reference UNIQUE constraint |

### Rollback Strategy

```sql
-- Remove new tables (reverse order)
DROP TABLE IF EXISTS purchase_draft_items CASCADE;
DROP TABLE IF EXISTS purchase_drafts CASCADE;
DROP TABLE IF EXISTS ocr_sessions CASCADE;
DROP TABLE IF EXISTS system_tutorials CASCADE;

-- Zero impact on existing tables
-- purchase_invoices, product_batches, stock_movements unchanged
```
