```ts
// =================================================================

interface PurchaseDraft {
  id: string;                          // UUID
  tenantId: string;

  // Source
  sourceType: DraftSource;             // enum — asal input

  // Supplier
  supplierId: string | null;
  supplierName: string;

  // Invoice
  invoiceNumber: string | null;        // null → auto-generate on confirm
  purchaseDate: string;                // ISO
  dueDate: string | null;

  // Items
  items: PurchaseDraftItem[];

  // Meta
  status: DraftStatus;                 // enum — current state
  confirmedBy: string | null;          // user ID
  confirmedAt: string | null;          // ISO

  // OCR linkage (future)
  ocrSessionId: string | null;

  createdAt: string;
  updatedAt: string;
}

// =================================================================
// PurchaseDraftItem — satu baris item dalam draft
// =================================================================

interface PurchaseDraftItem {
  id: string;                          // UUID

  // ——— Product Matching ———
  rawProductName: string;              // dari input (bisa beda dengan DB name)
  matchedProductId: string | null;     // setelah matching
  matchConfidence: number;             // 0–100
  matchMethod: "exact" | "fuzzy" | "manual" | "new" | "unmatched";

  // ——— Pricing (auto-loaded) ———
  previousPurchasePrice: number;       // auto — dari purchase history terakhir
  currentSellingPrice: number;         // auto — dari product.default_selling_price
  enteredPurchasePrice: number;        // from user / import

  // ——— Tax ———
  taxRate: number;                     // percent (e.g. 11)
  taxMode: "inclusive" | "exclusive";
  taxAmount: number;

  // ——— HPP ———
  hpp: number;                         // purchasePrice + taxAmount

  // ——— Margin + Recommendation ———
  targetMarginPercent: number;         // dari product master
  recommendedSellingPrice: number;     // HPP × (1 + margin)
  priceRecommendation: "use_recommended" | "keep_existing" | "manual";

  // ——— Batch ———
  batchNumber: string | null;          // null → auto
  expiredDate: string | null;

  // ——— Quantity ———
  quantity: number;
  unit: string;

  // ——— Status ———
  status: DraftItemStatus;
  warnings: string[];                  // "Price +15% dari terakhir"
  errorMessage: string | null;         // block confirmation if set
}

// =================================================================
// Enums
// =================================================================

type DraftSource = "manual" | "excel" | "csv" | "ocr";
type DraftStatus = "draft" | "review_required" | "confirmed" | "cancelled";
type DraftItemStatus = "pending" | "matched" | "priced" | "warning" | "error" | "confirmed";
```

### Status Machine

```
                  ┌─────────┐
                  │  DRAFT   │  ← initial state
                  └────┬─────┘
                       │ (all items matched + validated)
                       ▼
              ┌────────────────┐
```

```ts
    rawProductName: row.productName,
    matchedProductId: null,               // will be matched by ProductMatcher

    matchMethod: "unmatched",
    enteredPurchasePrice: row.purchasePrice ?? 0,
    quantity: row.qty ?? 1,
    batchNumber: row.batch ?? null,
    expiredDate: row.expiredDate ?? null,

    matchConfidence: item.aiConfidence ?? 0,
    matchMethod: "unmatched",
    enteredPurchasePrice: item.price ?? 0,
    quantity: item.qty ?? 1,
  }));
```

```ts
// New: purchase-draft auto-price engine
async function autoFillPricing(draftItem: PurchaseDraftItem): Promise<PurchaseDraftItem> {
  if (!draftItem.matchedProductId) return draftItem;

  // 1. Product master defaults (existing behavior)
  const product = await productRepo.getProductById(draftItem.matchedProductId);
  if (product) {
    draftItem.currentSellingPrice = product.defaultSellingPrice ?? 0;
    draftItem.targetMarginPercent = product.targetMarginPercent ?? 0;
    draftItem.unit = product.unit ?? "pcs";
  }

  // 2. Last purchase price (NEW — from purchase history)
  const lastPurchase = await getLastPurchasePrice(draftItem.matchedProductId);
  if (lastPurchase !== null) {
    draftItem.previousPurchasePrice = lastPurchase;
    if (draftItem.enteredPurchasePrice === 0) {
      draftItem.enteredPurchasePrice = lastPurchase;  // pre-fill if empty
    }
    // Warning if price changed significantly
    if (draftItem.enteredPurchasePrice > lastPurchase * 1.15) {
      draftItem.warnings.push(`Harga naik 15%+ dari pembelian terakhir (Rp ${lastPurchase})`);
    }
  }

  // 3. Last supplier for this product (NEW)
  const lastSupplier = await getLastSupplier(draftItem.matchedProductId);
  if (lastSupplier && !draftItem._supplierOverride) {
    // Suggest but don't override user's explicit supplier choice
  }

  return draftItem;
}

// Query helpers
async function getLastPurchasePrice(productId: string): Promise<number | null> {
  const { data } = await supabase
    .from("purchase_items")
    .select("unit_price")
    .eq("product_id", productId)
    .order("created_at", { ascending: false })
    .limit(1);
  return data?.[0]?.unit_price ?? null;
}
```

## STEP 6 — GLOBAL TAX ENGINE (DESIGN ONLY)

### Settings (stored in tenant/company settings)

```sql
-- Future migration (NOT NOW)
ALTER TABLE tenants ADD COLUMN purchase_tax_rate NUMERIC DEFAULT 0;
ALTER TABLE tenants ADD COLUMN purchase_tax_mode TEXT DEFAULT 'exclusive';
-- purchase_tax_mode: 'inclusive' | 'exclusive'
```

### Formula

```
EXCLUSIVE mode (default):
  HPP = PurchasePrice + (PurchasePrice × taxRate / 100)
  Example: Rp 15,000 + 11% = Rp 15,000 + Rp 1,650 = Rp 16,650

INCLUSIVE mode:
  HPP = PurchasePrice
  TaxAmount = PurchasePrice - (PurchasePrice / (1 + taxRate / 100))
  Example: Rp 15,000 (includes 11%) → Tax = Rp 1,486, HPP = Rp 15,000
```

### Tax Engine (pure function)

```ts
function calculateTax(
  purchasePrice: number,
  taxRate: number,        // 0–100 (percent)
  taxMode: "inclusive" | "exclusive"
): { taxAmount: number; priceBeforeTax: number; hpp: number } {
  if (taxRate === 0) {
    return { taxAmount: 0, priceBeforeTax: purchasePrice, hpp: purchasePrice };
  }

  if (taxMode === "inclusive") {
    const priceBeforeTax = purchasePrice / (1 + taxRate / 100);
    const taxAmount = purchasePrice - priceBeforeTax;
    return { taxAmount: Math.round(taxAmount), priceBeforeTax: Math.round(priceBeforeTax), hpp: purchasePrice };
  }

  // exclusive
  const taxAmount = purchasePrice * (taxRate / 100);
  return { taxAmount: Math.round(taxAmount), priceBeforeTax: purchasePrice, hpp: purchasePrice + Math.round(taxAmount) };
}
```
